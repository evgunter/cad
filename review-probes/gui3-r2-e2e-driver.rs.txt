//! TEMPORARY reviewer e2e driver — deleted before the review branch is
//! pushed; the transcript lives in review-probes/.
#![allow(clippy::expect_used, clippy::panic, clippy::unwrap_used, clippy::print_stdout)]

use pncad::document::SlotId;
use pncad::geom_core::Tol;
use viewer::props::SlotValue;
use viewer::session::{DocSession, Selection, SessionOp};
use viewer::{docio, tree};

fn show(label: &str, session: &DocSession) {
    println!(
        "  [{label}] busy={} gen={} landed={:?} states={} faults={}",
        session.busy(),
        session.generation().get(),
        session.landed_generation().map(viewer::evalseam::Generation::get),
        session.history().len(),
        tree::has_faults(&session.tree_rows())
    );
    for row in session.tree_rows() {
        println!(
            "      {:indent$}{} [{}] root={}{}",
            "",
            row.kind,
            row.status.badge(),
            row.root,
            row.status
                .message()
                .map(|m| format!("  — {m}"))
                .unwrap_or_default(),
            indent = row.depth * 2
        );
    }
}

#[test]
fn e2e_gallery_walk() {
    let tol = Tol::witness();
    let dir = std::path::PathBuf::from(
        std::env::var("R2_GALLERY").expect("set R2_GALLERY to a demo-tour gallery directory"),
    );

    // 1. Open a gallery document through the typed door.
    for name in ["ring", "checks", "heatsink", "diefillet"] {
        let path = dir.join(format!("{name}.pncad"));
        match docio::open(&path, tol) {
            Ok(history) => {
                let mut session = DocSession::inline(history.doc().clone(), tol);
                session.pump();
                println!("== {name}.pncad ==");
                show("opened", &session);
            }
            Err(e) => println!("== {name}.pncad == REFUSED: {e:?}"),
        }
    }

    // 2. The assembly workspace: InstantiatePart with no resolver.
    let asm = dir.join("assembly");
    let mut entries: Vec<_> = std::fs::read_dir(&asm)
        .expect("the assembly workspace exists")
        .filter_map(Result::ok)
        .map(|e| e.path())
        .filter(|p| p.extension().is_some_and(|x| x == "pncad"))
        .collect();
    entries.sort();
    for path in &entries {
        match docio::open(path, tol) {
            Ok(history) => {
                let mut session = DocSession::inline(history.doc().clone(), tol);
                session.pump();
                println!("== assembly/{} ==", path.file_name().unwrap().to_string_lossy());
                show("opened", &session);
            }
            Err(e) => println!(
                "== assembly/{} == REFUSED: {e:?}",
                path.file_name().unwrap().to_string_lossy()
            ),
        }
    }

    // 3. Walk the ring: edit, undo/redo across a branch, cancel, save.
    let mut session = DocSession::inline(
        docio::open(&dir.join("ring.pncad"), tol)
            .expect("ring opens")
            .doc()
            .clone(),
        tol,
    );
    session.pump();
    let revolve = session
        .tree_rows()
        .into_iter()
        .find(|r| r.kind == "Revolve")
        .expect("a revolve");
    session.perform(SessionOp::Select(Selection::Node(revolve.id)));
    println!("== ring: property rows for the revolve ==");
    for row in session.slot_rows() {
        println!(
            "      {:?} dim={:?} structural={} driver={:?} value={:?}",
            row.slot, row.dimension, row.structural, row.driver, row.value
        );
    }
    let angle = session
        .slot_rows()
        .into_iter()
        .find(|r| r.slot == SlotId::RevolveAngle)
        .expect("an angle slot");
    let outcome = session.perform(SessionOp::SetSlot {
        node: revolve.id,
        slot: SlotId::RevolveAngle,
        value: SlotValue::Continuous(angle.value.unwrap().as_f64() * 0.5),
    });
    println!("  half-turn edit: committed={} refusal={:?}", outcome.committed.len(), outcome.refusal);
    session.pump();
    show("after edit", &session);

    session.perform(SessionOp::Undo);
    let outcome = session.perform(SessionOp::SetSlot {
        node: revolve.id,
        slot: SlotId::RevolveAngle,
        value: SlotValue::Continuous(1.0),
    });
    println!("  branch edit: committed={} states={}", outcome.committed.len(), session.history().len());
    session.perform(SessionOp::Undo);
    session.perform(SessionOp::Redo);
    println!("  redo landed on states={} current={:?}", session.history().len(), session.history().current().index());

    // 4. Cancel an evaluation in flight.
    session.perform(SessionOp::SetSlot {
        node: revolve.id,
        slot: SlotId::RevolveAngle,
        value: SlotValue::Continuous(2.0),
    });
    println!("  before cancel:");
    show("pre-cancel", &session);
    session.perform(SessionOp::CancelEvaluation);
    session.pump();
    println!("  AFTER CancelEvaluation + pump:");
    show("post-cancel", &session);

    // 5. Save and re-open.
    let out = std::env::temp_dir().join("r2-e2e-ring.pncad");
    println!("  save: {:?}", session.perform(SessionOp::Save(out.clone())).refusal);
    let back = docio::open(&out, tol);
    println!("  reopen: {:?}", back.as_ref().map(|h| h.path_edits().len()));
}
