# Buckets two runs of `one_sum_differential` (module docs): base, then head.
#   python3 one_sum_differential.py.txt base.out head.out
import sys, collections

def load(p):
    d = {}
    for l in open(p):
        if l.startswith('P '):
            f = l.split()
            d[(f[1], f[2])] = (f[3], [float(x) for x in f[4:]])
    return d

ZERO = {'TangentLine', 'Line', 'TangentCircle', 'MeridianCircles', 'ApexTangentLine',
        'CoaxialCircles', 'CoincidentSurfaces', 'ERR:CoincidentSurfaces'}
SIGNED = {'ParallelLines': 1, 'Empty': -1}

def served(v):
    return not v.startswith('ESC:')

def against(v, t):
    if v in ZERO:
        return bool(t) and t[0] == t[0] and t[0] > 1.0
    if v in SIGNED and len(t) == 2:
        lo, hi = t
        return lo <= 1.0 if SIGNED[v] > 0 else hi >= -1.0
    return False

base, head = load(sys.argv[1]), load(sys.argv[2])
b = collections.defaultdict(collections.Counter)
for k in base:
    fam = k[0]
    (vb, tb), (vh, th) = base[k], head[k]
    c = b[fam]
    c['poses'] += 1
    if served(vb) and against(vb, tb): c['base served against truth'] += 1
    if served(vh) and against(vh, th): c['HEAD served against truth'] += 1
    if served(vh) and not served(vb): c['WIDENED (head serves, base escalated)'] += 1
    if served(vh) and served(vb) and vh != vb: c['FLIPPED'] += 1
    if served(vb) and not served(vh):
        c['head escalates, base was against' if against(vb, tb) else 'head escalates, base correct'] += 1
for fam in sorted(b):
    print(fam, dict(b[fam]))
